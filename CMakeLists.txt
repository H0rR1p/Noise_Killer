cmake_minimum_required(VERSION 3.20)
project(NoiseKiller VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(noisekiller_core STATIC src/core.cpp src/io.cpp)
target_include_directories(noisekiller_core PUBLIC include)
if(MSVC)
  target_compile_options(noisekiller_core PUBLIC /W4 /permissive- /utf-8 /EHsc)
endif()
add_executable(NoiseKiller.Cli src/cli.cpp)
target_link_libraries(NoiseKiller.Cli PRIVATE noisekiller_core)
if(WIN32)
  target_sources(NoiseKiller.Cli PRIVATE src/windows/profile.cpp src/windows/audio.cpp src/windows/render.cpp)
  target_link_libraries(NoiseKiller.Cli PRIVATE crypt32 advapi32 ole32 uuid avrt propsys)
  add_executable(NoiseKiller WIN32 src/windows/app.cpp src/windows/audio.cpp src/windows/profile.cpp src/windows/render.cpp)
  target_link_libraries(NoiseKiller PRIVATE noisekiller_core ole32 uuid avrt propsys crypt32 comctl32 advapi32 comdlg32)
  target_compile_definitions(NoiseKiller PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_definitions(NoiseKiller.Cli PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
enable_testing()
add_executable(noisekiller_tests tests/core_tests.cpp)
target_link_libraries(noisekiller_tests PRIVATE noisekiller_core)
if(WIN32)
  target_sources(noisekiller_tests PRIVATE src/windows/profile.cpp)
  target_link_libraries(noisekiller_tests PRIVATE crypt32 advapi32)
endif()
add_test(NAME noisekiller_contracts COMMAND noisekiller_tests)
add_executable(noisekiller_overlap_tests tests/mecag_overlap_tests.cpp)
target_link_libraries(noisekiller_overlap_tests PRIVATE noisekiller_core)
add_test(NAME noisekiller_synthetic_overlap_regression COMMAND noisekiller_overlap_tests)
set_tests_properties(noisekiller_synthetic_overlap_regression PROPERTIES TIMEOUT 60)
add_executable(noisekiller_holdout_tests tests/mecag_synthetic_holdout_tests.cpp)
target_link_libraries(noisekiller_holdout_tests PRIVATE noisekiller_core)
add_test(NAME noisekiller_synthetic_independent_holdout COMMAND noisekiller_holdout_tests)
set_tests_properties(noisekiller_synthetic_independent_holdout PROPERTIES TIMEOUT 60)
option(NOISEKILLER_BUILD_LEGACY_DRIVER_TESTS "Build superseded v0.4 self-built driver research tests (not current MVP)" OFF)
if(NOISEKILLER_BUILD_LEGACY_DRIVER_TESTS)
  add_library(noisekiller_driver_model STATIC src/driver/driver_model.cpp)
  target_include_directories(noisekiller_driver_model PUBLIC include)
  add_executable(noisekiller_driver_tests tests/driver_contract_tests.cpp)
  target_link_libraries(noisekiller_driver_tests PRIVATE noisekiller_driver_model)
  if(MSVC)
    target_compile_options(noisekiller_driver_model PUBLIC /W4 /permissive- /utf-8 /EHsc)
  endif()
  add_test(NAME noisekiller_legacy_driver_contract_model COMMAND noisekiller_driver_tests)
endif()
if(WIN32)
  add_executable(noisekiller_windows_tests tests/windows_contract_tests.cpp src/windows/audio.cpp src/windows/profile.cpp src/windows/render.cpp)
  target_link_libraries(noisekiller_windows_tests PRIVATE noisekiller_core ole32 uuid avrt propsys crypt32 advapi32)
  target_compile_definitions(noisekiller_windows_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_test(NAME noisekiller_windows_fail_closed COMMAND noisekiller_windows_tests)
  add_executable(noisekiller_render_tests tests/render_contract_tests.cpp)
  target_link_libraries(noisekiller_render_tests PRIVATE noisekiller_core)
  add_test(NAME noisekiller_cable_render_contracts COMMAND noisekiller_render_tests)
  add_test(NAME noisekiller_gui_no_capture_smoke COMMAND NoiseKiller --smoke)
  set_tests_properties(noisekiller_gui_no_capture_smoke PROPERTIES TIMEOUT 15)
endif()
install(TARGETS NoiseKiller.Cli RUNTIME DESTINATION .)
if(WIN32)
  install(TARGETS NoiseKiller RUNTIME DESTINATION .)
endif()
install(FILES README.md components.lock.json DESTINATION .)
# Detailed documents are local-only for now; a README-only clone still installs.
install(FILES NoiseKiller_需求规格说明书.md NoiseKiller_架构设计文档.md DESTINATION . OPTIONAL)
install(FILES docs/虚拟线路接入指南.md docs/MVP_构建与验收进度.md docs/MECAG_重叠修复与回归记录.md docs/MECAG_合成校准与独立留出.md DESTINATION docs OPTIONAL)
